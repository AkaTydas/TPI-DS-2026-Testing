    end

    subgraph S_ENTRY ["1. Entrada y Configuración Express"]
        INDEX["src/index.ts

        ENV["src/config/env.ts

        DB["src/config/database.ts
(Pool de PostgreSQL y conexión)"]
    end

        AUTH["src/middlewares/auth.ts
(Verifica JWT y valida roles de usuario)"]
        ERR_HANDLER["src/middlewares/errorHandler.ts
(Captura errores y formatea RFC 7807)"]
        NOT_FOUND["src/middlewares/notFoundHandler.ts
(Rutas 404 no encontradas)"]
    end

    subgraph S_ROUTES ["3. Rutas de la Aplicación"]

        R_HEALTH["src/routes/healthRoutes.ts
(/health)"]
        R_DOCS["src/routes/docRoutes.ts
(/docs y /openapi.yaml)"]
    end

    subgraph S_CONTROLLERS ["4. Controladores (Validación Zod y Despacho)"]
        C_RES["src/controllers/reservationController.ts

        C_HLT["src/controllers/healthController.ts

    end

    subgraph S_SERVICES ["5. Servicios de Negocio (Lógica Central)"]

    end

        CLI_M4["src/clients/m4Client.ts
